5. TOP 5 HIDDEN ARCHITECTURAL BLINDSPOTS & REMEDIATIONS
================================================================================

1. Stateless Polyphase FIR Resampling Boundary Notch (`tts.py`):
   - Issue: `resample_poly` zero-pads FIR filter delay lines per chunk, causing sub-millisecond amplitude dips.
   - Fix: Maintain a 32-sample lookbehind overlap buffer (`_resample_overlap_tail`) across chunks.

2. Win32 `AttachThreadInput` Synchronous Thread Freeze on Hung Windows (`os_control.py`):
   - Issue: Attaching input to a frozen foreground window hangs the Jarvis thread indefinitely.
   - Fix: Probe `user32.IsHungAppWindow(fg_hwnd)` first; fallback to synthetic `VK_MENU` (`Alt`) key events.

3. `DWMWA_CLOAKED` Virtual Desktop False-Positive Blackout (`privacy_guard.py`):
   - Issue: Windows 11 cloaks windows on inactive virtual desktops while keeping `IsWindowVisible = True`.
   - Fix: Query `dwmapi.DwmGetWindowAttribute(hwnd, 14, ...)` to skip cloaked windows.

4. UTF-8 Multibyte Split Across `ReadFile` Boundaries (`antigravity_bridge.py`):
   - Issue: Decoding chunked stdout bytes directly splits multibyte characters, causing `UnicodeDecodeError` or ``.
   - Fix: Use `codecs.getincrementaldecoder("utf-8")(errors="replace")` on the stream reader instance.

5. Unclosed Read-Transaction WAL Tail Pinning (`ro_pool.py`):
   - Issue: Borrowed `?mode=ro` connections left with open read transactions prevent WAL truncation.
   - Fix: Explicitly release read snapshots (`conn.commit()` / `isolation_level = None`) in the `finally` block.

6.1 Stateful Overlap-Save 48kHz Polyphase Resampler (`jarvis/audio/tts.py`)
--------------------------------------------------------------------------------
```python
from __future__ import annotations
import numpy as np
from scipy.signal import resample_poly

class StatefulPolyphaseUpsampler:
    __slots__ = ("_up", "_down", "_pad_samples", "_tail_24k")
    def __init__(self, up: int = 2, down: int = 1, pad_samples: int = 32) -> None:
        self._up = int(up)
        self._down = int(down)
        self._pad_samples = int(pad_samples)
        self._tail_24k: np.ndarray | None = None

    def reset(self) -> None:
        self._tail_24k = None

    def process(self, chunk_24k: np.ndarray) -> np.ndarray:
        pcm = np.asarray(chunk_24k, dtype=np.float32).ravel()
        if pcm.size == 0 or self._up == self._down:
            return pcm
        p = self._pad_samples
        if self._tail_24k is not None and self._tail_24k.size == p:
            extended = np.concatenate((self._tail_24k, pcm))
            upsampled = resample_poly(extended, up=self._up, down=self._down).astype(np.float32, copy=False)
            trim = (p * self._up) // self._down
            out = upsampled[trim:]
        else:
            out = resample_poly(pcm, up=self._up, down=self._down).astype(np.float32, copy=False)
        self._tail_24k = pcm[-p:].copy() if pcm.size >= p else None
        return np.ascontiguousarray(out, dtype=np.float32)
```

6.2 DWM Cloak-Aware Z-Order Probe (`jarvis/vision/privacy_guard.py`)
--------------------------------------------------------------------------------
```python
from __future__ import annotations
import ctypes
from ctypes import wintypes
import sys

IS_WINDOWS = sys.platform == "win32"
DWMWA_CLOAKED = 14

def is_window_dwm_cloaked(hwnd: int) -> bool:
    if not IS_WINDOWS or not hwnd:
        return False
    try:
        dwmapi = ctypes.windll.dwmapi
        cloaked = wintypes.DWORD(0)
        hr = dwmapi.DwmGetWindowAttribute(
            wintypes.HWND(hwnd), wintypes.DWORD(DWMWA_CLOAKED),
            ctypes.byref(cloaked), ctypes.sizeof(cloaked)
        )
        return hr == 0 and bool(cloaked.value != 0)
    except Exception:
        return False
```

6.3 Hung-Window-Safe Foreground Focus Activator (`jarvis/skills/os_control.py`)
--------------------------------------------------------------------------------
```python
from __future__ import annotations
import ctypes
from ctypes import wintypes
import sys

IS_WINDOWS = sys.platform == "win32"
SW_RESTORE = 9
VK_MENU = 0x12
KEYEVENTF_EXTENDEDKEY = 0x0001
KEYEVENTF_KEYUP = 0x0002

def safe_set_foreground_window(target_hwnd: int) -> bool:
    if not IS_WINDOWS or not target_hwnd:
        return False
    user32 = ctypes.windll.user32
    kernel32 = ctypes.windll.kernel32
    hwnd = wintypes.HWND(target_hwnd)
    if user32.IsIconic(hwnd):
        user32.ShowWindow(hwnd, SW_RESTORE)
    fg_hwnd = user32.GetForegroundWindow()
    if fg_hwnd == target_hwnd:
        return True
    fg_is_hung = bool(fg_hwnd and user32.IsHungAppWindow(wintypes.HWND(fg_hwnd)))
    if not fg_is_hung and fg_hwnd:
        curr_tid = kernel32.GetCurrentThreadId()
        fg_tid = user32.GetWindowThreadProcessId(wintypes.HWND(fg_hwnd), None)
        if curr_tid and fg_tid and curr_tid != fg_tid:
            attached = bool(user32.AttachThreadInput(curr_tid, fg_tid, True))
            try:
                user32.BringWindowToTop(hwnd)
                if user32.SetForegroundWindow(hwnd):
                    return True
            finally:
                if attached:
                    user32.AttachThreadInput(curr_tid, fg_tid, False)
    user32.keybd_event(VK_MENU, 0, KEYEVENTF_EXTENDEDKEY, 0)
    user32.keybd_event(VK_MENU, 0, KEYEVENTF_EXTENDEDKEY | KEYEVENTF_KEYUP, 0)
    return bool(user32.SetForegroundWindow(hwnd))
```

```python
from __future__ import annotations
import argparse, ast, sys
from pathlib import Path

def build_ast_skeleton(file_path: Path) -> str:
    source = file_path.read_text(encoding="utf-8", errors="replace")
    tree = ast.parse(source, filename=str(file_path))
    lines: list[str] = [f"# AST SKELETON: {file_path.as_posix()}"]
    for node in tree.body:
        if isinstance(node, ast.ClassDef):
            lines.append(f"\nclass {node.name}:")
            for item in node.body:
                if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    lines.append(f"    def {item.name}(...)")
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            lines.append(f"\ndef {node.name}(...)")
    return "\n".join(lines)
```
